# Application build targets

if(WITH_CYCLES_STANDALONE)
  file( GLOB SRC *.cpp *.h *.c)

  # Windows resources: VERSIONINFO and the side-by-side private assembly
  # manifest (see ccycles_manifest.xml.in) are compiled into ccycles.dll.
  if(WIN32)
    # Private assembly members: ccycles plus every DLL Rhino ships its own copy
    # of, named as installed. A wrong name silently loses isolation for that DLL,
    # so recheck against <install>/Plug-ins after a library bundle update.
    # fix-cycles-sxs.ps1 reads this list back from the built manifest.
    set(CCYCLES_SXS_FILES
      ccycles.dll
      OpenImageIO.dll
      OpenImageIO_Util.dll
      OpenColorIO_2_5.dll
      openvdb.dll
      tbb12.dll
      tbbmalloc.dll
      tbbmalloc_proxy.dll
    )

    if(WITH_CYCLES_DEVICE_ONEAPI)
      list(APPEND CCYCLES_SXS_FILES cycles_kernel_oneapi_jit.dll sycl8.dll)
    endif()

    list(REMOVE_DUPLICATES CCYCLES_SXS_FILES)

    set(CCYCLES_ASSEMBLY_VERSION "1.0.0.0" CACHE STRING
      "assemblyIdentity version stamped into the ccycles SxS manifest")

    set(CCYCLES_SXS_FILE_ELEMENTS "")
    foreach(_sxs_file ${CCYCLES_SXS_FILES})
      string(APPEND CCYCLES_SXS_FILE_ELEMENTS "  <file name=\"${_sxs_file}\"/>\n")
    endforeach()
    unset(_sxs_file)

    configure_file(
      ${CMAKE_CURRENT_SOURCE_DIR}/ccycles_manifest.xml.in
      ${CMAKE_CURRENT_BINARY_DIR}/ccycles.manifest
      @ONLY
    )

    # Generated .rc embedding the manifest as type 24, ID 2
    # (ISOLATIONAWARE_MANIFEST_RESOURCE_ID), which the loader reads for a DLL.
    # Absolute path, so RC needs no -I. Staged through configure_file(COPYONLY) so
    # reconfiguring (build_cycles.ps1 does, every run) does not relink ccycles.dll.
    file(TO_NATIVE_PATH "${CMAKE_CURRENT_BINARY_DIR}/ccycles.manifest" _ccycles_manifest_path)
    string(REPLACE "\\" "\\\\" _ccycles_manifest_path "${_ccycles_manifest_path}")
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/ccycles_manifest.rc.staging
      "// Generated by CMake - do not edit.\n2 24 \"${_ccycles_manifest_path}\"\n")
    configure_file(
      ${CMAKE_CURRENT_BINARY_DIR}/ccycles_manifest.rc.staging
      ${CMAKE_CURRENT_BINARY_DIR}/ccycles_manifest.rc
      COPYONLY
    )
    unset(_ccycles_manifest_path)

    # ccycles_version.rc uses Rhino's generated src4/version.h, absent in a
    # standalone checkout. Default to the submodule layout, allow an override,
    # and fall back to the version.h beside the .rc rather than fail the build.
    set(RHINO_VERSION_H "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../version.h"
        CACHE FILEPATH "Rhino's generated version.h, stamped into ccycles.dll")

    if(NOT EXISTS "${RHINO_VERSION_H}")
      message(STATUS "ccycles: ${RHINO_VERSION_H} not found, version resource falls back to src/ccycles/version.h")
      set_property(SOURCE ccycles_version.rc APPEND PROPERTY COMPILE_DEFINITIONS STANDALONE)
    endif()

    list(APPEND SRC
      ${CMAKE_CURRENT_SOURCE_DIR}/ccycles_version.rc
      ${CMAKE_CURRENT_BINARY_DIR}/ccycles_manifest.rc
    )
  endif()


  add_library(ccycles SHARED ${SRC} ${INC} ${INC_SYS})

  if(WIN32)
    # The linker's own manifest would collide with resource 2 embedded above.
    set_property(TARGET ccycles APPEND_STRING PROPERTY LINK_FLAGS " /MANIFEST:NO")
  endif()
  unset(SRC)

  target_link_libraries(ccycles PRIVATE ${LIB})

  if(APPLE)
    if(WITH_CYCLES_STANDALONE_GUI)
      set_property(TARGET ccycles APPEND_STRING PROPERTY LINK_FLAGS
        " -framework Cocoa -framework CoreAudio -framework AudioUnit -framework AudioToolbox -framework ForceFeedback -framework CoreVideo")
    endif()
  endif()

  if(UNIX AND NOT APPLE)
    set_target_properties(ccycles PROPERTIES INSTALL_RPATH $ORIGIN/lib)
  endif()

  if(CYCLES_STANDALONE_REPOSITORY)
    cycles_install_libraries(ccycles)
  endif()

  if(WITH_USD AND USD_LIBRARY_DIR)
    install(DIRECTORY
      ${USD_LIBRARY_DIR}/usd
      DESTINATION ${CMAKE_INSTALL_PREFIX}
    )

    install(DIRECTORY
      ${USD_LIBRARY_DIR}/../plugin/usd
      DESTINATION ${CMAKE_INSTALL_PREFIX}
    )
  endif()

  install(PROGRAMS
    $<TARGET_FILE:ccycles>
    DESTINATION ${CMAKE_INSTALL_PREFIX})
endif()
